Google Sheets integration

Submissions are automatically pushed to a Google Sheet. Paste your Apps Script Web App URL below.

How to set up Google Sheets
  1. Open sheets.new to create a spreadsheet.
  2. Menu: Extensions → Apps Script.
  3. Delete any sample code and paste the snippet below.
  4. Click Deploy → New deployment → Web app.
  5. Set Execute as: Me and Who has access: Anyone.
  6. Copy the Web app URL and paste it above.
function doPost(e) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(['Date', 'Name', 'Phone', 'Email']);
  }
  const d = JSON.parse(e.postData.contents);
  sheet.appendRow([d.created_at || new Date(), d.name, d.phone, d.email]);
  return ContentService.createTextOutput(JSON.stringify({ ok: true }))
    .setMimeType(ContentService.MimeType.JSON);
}
